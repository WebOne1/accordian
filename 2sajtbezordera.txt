  Otključaj lekciju
</button>
                        </div>
                      )}
                    </div>
                  );
                })}
            </div>
          )}
        </div>
      </section>

      {/* 3. SEKCIJA: POJEDINAČNE LEKCIJE (Single Purchase) */}
      <section id="single-lessons" className="px-6 md:px-16 py-22 max-w-7xl mx-auto">
        <div className="text-center md:text-left mb-12">
          <span className="text-blue-500 text-xs font-bold uppercase tracking-widest block mb-2">Zasebna kupovina</span>
          <h2 className="text-4xl font-black uppercase tracking-tight">Pojedinačne Masterclass Lekcije</h2>
          <p className="text-gray-500 dark:text-gray-400 mt-2 text-sm max-w-xl font-light">
            Specijalni raspisani aranžmani i namenske solo deonice koje možete kupiti zasebno, bez mesečne obaveze.
          </p>
        </div>

        {loading ? (
          <p className="text-center text-gray-500 py-10">Učitavanje lekcija...</p>
        ) : (
          <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-8">
            {videos
              .filter((video) => video.access_type === "single_purchase")
              .map((video) => {
                const hasPurchased = purchasedVideos.includes(video.id);

                return (
                  <div
                    key={video.id}
                    className={`
group
rounded-3xl
overflow-hidden
border
border-2
border-white/10
bg-gradient-to-b
from-[#151515]
to-[#0f0f0f]
flex
flex-col
justify-between
transition-all
duration-500
hover:-translate-y-2
hover:border-blue-500/50
hover:shadow-[0_0_40px_rgba(59,130,246,0.15)]
`}
                  >
                    {hasPurchased ? (
                      <>
                        <HlsPlayer 
                          lekcijaSlug={video.title.toLowerCase().replace(/\s+/g, "")} 
                          userId={userId} 
                        />

                        <div className="p-5">
                          <span className="bg-emerald-600/10 text-emerald-500 px-3 py-1 text-xs font-bold rounded-full tracking-wider uppercase">
                            Kupljeno / Pristup omogućen
                          </span>
                          <h3 className="mt-3 text-lg font-bold tracking-tight">
                            {video.title}
                          </h3>
                        </div>
                      </>
                    ) : (
                      /* Prikaz za kupovinu */
                      <div className="p-8 text-center flex flex-col justify-between items-center h-full min-h-[260px]">
                        <div className="w-12 h-12 bg-blue-500/10 rounded-full flex items-center justify-center text-xl text-blue-500">
                          🎵
                        </div>
                        <div className="space-y-2">
                          <span className="bg-blue-600/10 text-blue-500 px-3 py-1 rounded-full text-[10px] font-black tracking-widest uppercase">
                            Single Purchase
                          </span>
                          <h3 className="text-lg font-bold tracking-tight pt-1">
                            {video.title}
                          </h3>
                          <p className="text-gray-400 text-xs font-light max-w-xs">
                            Otključajte trajni pristup ovoj lekciji i pratećim materijalima.
                          </p>
                        </div>
                        <button
  className="
  w-full
  mt-6
  bg-gradient-to-r
  from-blue-600
  to-blue-500
  text-white
  py-4
  rounded-xl
  font-black
  uppercase
  tracking-wider
  hover:scale-[1.03]
  transition
"
>
  Kupi pristup
</button>
                      </div>
                    )}
                  </div>
                );
              })}
          </div>
        )}
      </section></>
)}
      
      {/* ========================================================================= */}
      {/* LOGIN POP-UP WINDOW (MODAL) */}
      {/* ========================================================================= */}
      {prikaziLogin && (
        <div className="fixed inset-0 z-[99] flex items-center justify-center bg-black/60 backdrop-blur-sm p-4 fade-in">
          {/* Pozadina iza prozora koja takođe zatvara pop-up na klik sa strane */}
          <div className="absolute inset-0" onClick={() => setPrikaziLogin(false)} />

          {/* Sadržaj prozora */}
          <div
            className={`relative z-10 w-full max-w-md rounded-2xl p-8 shadow-2xl transition-colors duration-500 border ${
              darkMode 
                ? "bg-[#111] text-white border-zinc-800" 
                : "bg-white text-black border-gray-100"
            }`}
          >
            {/* Dugme za zatvaranje (X) u gornjem desnom uglu */}
            <button
              onClick={() => setPrikaziLogin(false)}
              className="absolute top-4 right-4 text-xl font-light hover:scale-110 transition opacity-60 hover:opacity-100"
            >
              ✕
            </button>

            {/* Naslov pop-up prozora */}
            <div className="text-center mb-6">
              <h3 className="text-2xl font-black uppercase tracking-tight">Prijavi se</h3>
              <p className="text-xs text-gray-400 mt-1">Pristupite vašim lekcijama za harmoniku</p>

              {authError && (
    <div className="mt-4 bg-red-500/10 border border-red-500/30 text-red-400 text-sm rounded-lg px-4 py-3">
      {authError}
    </div>
  )}

            </div>

            {/* Login Forma */}
            <form
  className="space-y-4"
  onSubmit={(e) => {
    e.preventDefault();
    signIn();
  }}
>
              <div>
                <label className="block text-xs font-bold uppercase tracking-wider text-gray-400 mb-1.5">
                  Email adresa
                </label>
                <input
                  type="email"
  placeholder="ime@primer.com"
  value={email}
  onChange={(e) => setEmail(e.target.value)}
                  className={`w-full px-4 py-3 rounded-lg border text-sm focus:outline-none focus:ring-2 focus:ring-[#9f6b3d] transition ${
                    darkMode
                      ? "bg-zinc-900 border-zinc-800 text-white placeholder-gray-600"
                      : "bg-gray-50 border-gray-200 text-black placeholder-gray-400"
                  }`}
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase tracking-wider text-gray-400 mb-1.5">
                  Lozinka
                </label>
                <input
  type="password"
  placeholder="••••••••"
  value={password}
  onChange={(e) => setPassword(e.target.value)}
                  className={`w-full px-4 py-3 rounded-lg border text-sm focus:outline-none focus:ring-2 focus:ring-[#9f6b3d] transition ${
                    darkMode
                      ? "bg-zinc-900 border-zinc-800 text-white placeholder-gray-600"
                      : "bg-gray-50 border-gray-200 text-black placeholder-gray-400"
                  }`}
                />
              </div>

              {/* Dugme za slanje unutar pop-upa */}
              <button
  type="submit"
  className="w-full bg-[#9f6b3d] hover:bg-[#855830] text-white font-bold uppercase text-xs tracking-widest py-3.5 rounded-lg shadow-lg hover:scale-[1.01] transition mt-2"
>
  Uloguj se
</button>
            </form>

            {/* Dodatne opcije na dnu */}
            <div className="text-center mt-6 text-xs text-gray-400 space-y-2">
              <a href="#" className="hover:underline block">Zaboravili ste lozinku?</a>
              <p>
                Nemate nalog?{" "}
                <button
  onClick={signUp}
  className="text-[#9f6b3d] font-bold hover:underline"
>
  Registrujte se
</button>
              </p>
            </div>
          </div>
        </div>
      )}
      {/* ========================================================================= */}
    </main>
  );
}
